# Create or update session polls.

PUT {{baseUrl}}/zoom_events/events/:eventId/sessions/:sessionId/polls


<p>
Use this API to create or update the list of polls for the session.
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
zoom_events_sessions:write
</code>

<code>
zoom_events_sessions:write:admin
</code>

</p>



<p>

<strong>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">
Rate Limit Label
</a>
:
</strong>
 
<code>
Light
</code>

</p>
