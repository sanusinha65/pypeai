# Create a webinar&#39;s branding name tag

POST {{baseUrl}}/webinars/:webinarId/branding/name_tags


<p>
Use this API to create a webinar&#39;s 
<a href="https://support.zoom.us/hc/en-us/articles/4836268732045-Using-Webinar-Session-Branding">
Session Branding
</a>
 name tag. There&#39;s a limit of 20 name tags per webinar. 
<strong>
Prerequisites:
</strong>
 
*  The 
<strong>
Webinar Session Branding
</strong>
 setting enabled.
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
webinar:write
</code>
,
<code>
webinar:write:admin
</code>

</p>



<p>

<strong>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">
Rate Limit Label
</a>
:
</strong>
 
<code>
Medium
</code>

</p>
