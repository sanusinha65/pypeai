# Get recording by call ID

GET {{baseUrl}}/phone/call_logs/:id/recordings


<p>
Returns an account&#39;s 
<a href="https://support.zoom.us/hc/en-us/articles/360038521091-Accessing-and-sharing-call-recordings">
call recording
</a>
 by the recording&#39;s 
<code>
callId
</code>
 or 
<code>
callLogId
</code>
 ID. 
</p>



<p>

<strong>
Note
</strong>
: This  API returns the 
<code>
file_url
</code>
 in the JSON query results.
</p>



<p>

<strong>
Prerequisites:
</strong>
 
* A Pro or higher account with Zoom Phone license 
* Account owner or admin privileges
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
phone:read:admin
</code>
,
<code>
phone_recording:read
</code>
,
<code>
phone_recording:read:admin
</code>

</p>



<p>

<strong>
Granular Scopes:
</strong>
 
<code>
phone:read:call_recording
</code>
,
<code>
phone:read:call_recording:admin
</code>

</p>



<p>

<strong>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">
Rate Limit Label
</a>
:
</strong>
 
<code>
LIGHT
</code>

</p>
