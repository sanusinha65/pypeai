# Get device line keys information

GET {{baseUrl}}/phone/devices/:deviceId/line_keys


<p>
Use this API to get information on the Zoom Phone device 
<a href="https://support.zoom.us/hc/en-us/articles/4402415568397-Customizing-keys-for-devices-with-multiple-users">
line keys
</a>
 settings and position.
</p>



<p>

<strong>
Prerequisites:
</strong>
 
* A Business or Enterprise account 
* A Zoom Phone license
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
phone:read:admin
</code>
,
<code>
phone:read
</code>

</p>



<p>

<strong>
Granular Scopes:
</strong>
 
<code>
phone:read:device_line_keys
</code>
,
<code>
phone:read:device_line_keys:admin
</code>

</p>



<p>

<strong>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">
Rate Limit Label
</a>
:
</strong>
 
<code>
Heavy
</code>

</p>
