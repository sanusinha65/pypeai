# Get a provision template

GET {{baseUrl}}/phone/provision_templates/:templateId


<p>
Use this API to get a specific 
<a href="https://support.zoom.us/hc/en-us/articles/360035817952">
provision template
</a>
.
</p>



<p>

<strong>
Prerequisites:
</strong>
 
* A Pro or higher account plan 
* A Zoom Phone license
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
phone:read:admin
</code>

</p>



<p>

<strong>
Granular Scopes:
</strong>
 
<code>
phone:read:provision_template:admin
</code>

</p>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">

</a>

</code>

</p>
