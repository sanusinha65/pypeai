# Remove a calling plan from a Zoom Room

DELETE {{baseUrl}}/phone/rooms/:roomId/calling_plans/:type?billing_account_id=&lt;string&gt;


<p>
Use this API to unassign a 
<a href="https://marketplace.zoom.us/docs/api-reference/other-references/plans#zoom-phone-calling-plans">
calling plan
</a>
 from a 
<a href="https://support.zoom.us/hc/en-us/articles/360025153711#h_140e30ba-5a88-40b9-b799-16883fa0a037">
Zoom Room
</a>
.
</p>



<p>

<strong>
Prerequisites:
</strong>
 
* A Business or Enterprise account 
* A Zoom Phone license
</p>

<strong>
Scopes:

<code>
phone:write:admin
</code>

</p>

<strong>

</p>
