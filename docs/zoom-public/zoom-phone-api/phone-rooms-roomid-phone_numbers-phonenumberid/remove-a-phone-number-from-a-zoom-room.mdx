# Remove a phone number from a Zoom Room

DELETE {{baseUrl}}/phone/rooms/:roomId/phone_numbers/:phoneNumberId


<p>
Use this API to unassign a 
<a href="https://support.zoom.us/hc/en-us/articles/360020808292-Managing-Phone-Numbers#h_38ba8b01-26e3-4b1b-a9b5-0717c00a7ca6">
phone number
</a>
 from a 
<a href="https://support.zoom.us/hc/en-us/articles/360025153711#h_140e30ba-5a88-40b9-b799-16883fa0a037">
Zoom Room
</a>
.
</p>



<p>

<strong>
Prerequisites:
</strong>
 
* A Business or Enterprise account 
* A Zoom Phone license 
* The Zoom Room must have been previously assigned a Zoom Phone number
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
phone:write:admin
</code>

</p>



<p>

<strong>
Granular Scopes:
</strong>
 
<code>
phone:delete:room_phone_number:admin
</code>

</p>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">

</a>
:

</code>

</p>
