# Get a shared line group policy

GET {{baseUrl}}/phone/shared_line_groups/:sharedLineGroupId/policies


<p>
Returns the policy setting of a specific 
<a href="https://support.zoom.us/hc/en-us/articles/360038850792">
shared line group
</a>
.
</p>



<p>

<strong>
Prerequisites:
</strong>

* Pro or a higher account with Zoom Phone license
* Account owner or admin permissions
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
phone:read:admin
</code>

</p>



<p>

<strong>
Granular Scopes:

<code>
phone:read:shared_line_group_policy:admin
</code>

</p>

</a>

LIGHT
</code>

</p>
