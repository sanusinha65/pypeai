# Add users to a directory of a site

POST {{baseUrl}}/phone/sites/:siteId/dial_by_name_directory/extensions


<p>
Use this API to add users to a 
<a href="https://support.zoom.us/hc/en-us/articles/4404938949389-Using-a-dial-by-name-directory">
directory
</a>
 of the specified site.
</p>



<p>

<strong>
Prerequisites:
</strong>

* Business or Education account
* Zoom Phone license 
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
phone:write:admin
</code>

</p>



<p>

<strong>
Granular Scopes:
</strong>
 
<code>
phone:write:directory:admin
</code>

</p>



<p>

<strong>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">

</a>

Light
</code>

</p>
