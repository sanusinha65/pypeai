# Get a user&#39;s profile settings

GET {{baseUrl}}/phone/users/:userId/settings


<p>
Gets the Zoom Phone 
<a href="https://support.zoom.us/hc/en-us/articles/360021325712-Configuring-Settings">
profile settings
</a>
 of a user. For user-level apps, pass 
<a href="https://marketplace.zoom.us/docs/api-reference/using-zoom-apis#mekeyword">
the 
<code>
me
</code>
 value
</a>
 instead of the 
<code>
userId
</code>
 parameter.
</p>



<p>

<strong>
Prerequisites:
</strong>
 
* A Business or Enterprise account 
* A Zoom Phone license
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
phone:read:admin
</code>
,
<code>
phone:read
</code>

</p>



<p>

<strong>
Granular Scopes:
</strong>
 
<code>
phone:read:user_setting:admin
</code>
,
<code>
phone:read:user_setting
</code>

</p>



<p>

<strong>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">
Rate Limit Label
</a>
:
</strong>
 
<code>
Light
</code>

</p>
