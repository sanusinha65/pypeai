# Delete Virtual Background files

DELETE {{baseUrl}}/groups/:groupId/settings/virtual_backgrounds?file_ids=&lt;string&gt;


<p>
Use this API to delete a group&#39;s 
<a href="https://support.zoom.us/hc/en-us/articles/210707503-Virtual-Background#h_01EJF3YFEWGT8YA0ZJ079JEDQE">
Virtual Background files
</a>
. 
</p>



<p>

<strong>
Prerequisites:
</strong>
 
* The 
<a href="https://support.zoom.us/hc/en-us/articles/210707503-Virtual-Background#h_2ef28080-fce9-4ac2-b567-dc958afab1b7">
Virtual Background feature
</a>
 must be enabled on the account.
</p>



<p>

<strong>
Scopes:
</strong>
 
<code>
group:write:admin
</code>

</p>



<p>

<strong>

<a href="https://marketplace.zoom.us/docs/api-reference/rate-limits#rate-limits">
Rate Limit Label
</a>
:
</strong>
 
<code>
LIGHT
</code>

</p>
